```python
    # Navigation (Proceed to Next Step)
    if st.session_state.trained_model is not None:
        st.divider()
        
        if st.button("Proceed to Next Step", type="primary"):
            # Auto-save checkpoint (D-13)
            try:
                # Extract dataset name from session state (set in Phase 06)
                dataset_name = st.session_state.uploaded.name if st.session_state.uploaded else "unknown"
                # Remove file extension
                if '.' in dataset_name:
                    dataset_name = dataset_name.rsplit('.', 1)[0]
                
                # Prepare metadata
                metadata = {
                    'timestamp': datetime.now().isoformat(),
                    'dataset_name': dataset_name,
                    'topic_count': len(st.session_state.trained_model.get_topics()) - 1,  # Exclude outlier topic -1
                    'outlier_count': len([t for t in st.session_state.topics if t == -1]),
                    'total_documents': len(st.session_state.topics),
                    'outlier_percentage': (len([t for t in st.session_state.topics if t == -1]) / len(st.session_state.topics)) * 100
                }
                
                # Save checkpoint (D-16: timestamp + dataset name)
                with st.spinner("Saving checkpoint..."):
                    checkpoint_dir = save_checkpoint(
                        config=st.session_state.model_config,
                        metadata=metadata,
                        model=st.session_state.trained_model,
                        embeddings=st.session_state.embeddings,
                        dataset_fingerprint=st.session_state.dataset_fingerprint
                    )
                
                st.success(f"✓ Checkpoint saved: {checkpoint_dir.name}")
                
                # Update available checkpoints list for Phase 07 UI
                from checkpoint_manager import list_checkpoints
                st.session_state.available_checkpoints = list_checkpoints()
                
                # Advance to Step 4 (Phase 09: Basic Visualization)
                goto(4)
                
            except Exception as e:
                # Block navigation on failure (D-14)
                st.error(f"❌ Checkpoint save failed: {e}")
                st.error("Cannot proceed to next step until checkpoint is saved. Please resolve the issue (disk space, permissions) and try again.")
    else:
        # Show message if no model trained yet
        st.info("Train a model to proceed to the next step")
```

Note: Per D-15, there is no manual "Save Checkpoint Now" button. Checkpoint is only saved automatically on "Proceed to Next Step" click.

    # Navigation Guard Prompts (D-20: in sidebar near navigation)
    if st.session_state.navigation_pending is not None:
        target_step = st.session_state.navigation_pending
        
        st.divider()
        
        # If model trained: offer to save checkpoint (D-18)
        if st.session_state.get('trained_model') is not None:
            st.warning("⚠️ You have a trained model. Save checkpoint before going back?")
            
            col1, col2, col3 = st.columns(3)
            
            with col1:
                if st.button("Save & Go", key="guard_save_go", use_container_width=True):
                    # Save checkpoint
                    try:
                        from checkpoint_manager import save_checkpoint, list_checkpoints
                        from datetime import datetime
                        
                        dataset_name = st.session_state.uploaded.name if st.session_state.uploaded else "unknown"
                        if '.' in dataset_name:
                            dataset_name = dataset_name.rsplit('.', 1)[0]
                        
                        metadata = {
                            'timestamp': datetime.now().isoformat(),
                            'dataset_name': dataset_name,
                            'topic_count': len(st.session_state.trained_model.get_topics()) - 1,
                            'outlier_count': len([t for t in st.session_state.topics if t == -1]),
                            'total_documents': len(st.session_state.topics),
                            'outlier_percentage': (len([t for t in st.session_state.topics if t == -1]) / len(st.session_state.topics)) * 100
                        }
                        
                        checkpoint_dir = save_checkpoint(
                            config=st.session_state.model_config,
                            metadata=metadata,
                            model=st.session_state.trained_model,
                            embeddings=st.session_state.embeddings,
                            dataset_fingerprint=st.session_state.dataset_fingerprint
                        )
                        
                        st.session_state.available_checkpoints = list_checkpoints()
                        st.success(f"✓ Saved: {checkpoint_dir.name}")
                        
                        # Navigate after successful save
                        st.session_state.navigation_pending = None
                        goto(target_step)
                        
                    except Exception as e:
                        st.error(f"Save failed: {e}")
            
            with col2:
                if st.button("Go Back", key="guard_go_nosave", use_container_width=True):
                    st.session_state.navigation_pending = None
                    goto(target_step)
            
            with col3:
                if st.button("Cancel", key="guard_cancel", use_container_width=True):
                    st.session_state.navigation_pending = None
                    st.rerun()
        
        # If NO model trained: confirm parameter loss (D-17)
        else:
            st.warning("⚠️ Are you sure? Parameter changes will be lost.")
            
            col1, col2 = st.columns(2)
            
            with col1:
                if st.button("Go Back", key="guard_confirm_go", use_container_width=True):
                    st.session_state.navigation_pending = None
                    goto(target_step)

                    st.session_state.navigation_pending = None
